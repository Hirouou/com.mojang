  // Handwheels: many degrees of handwheel movement translate to a small change in gun angle.
  function bindWheel(el,kind){
    let id=null,lastA=0,visual=0;const arm=$(kind==='az'?'azArm':'elArm');
    el.addEventListener('pointerdown',e=>{e.preventDefault();e.stopPropagation();touchUI();id=e.pointerId;el.setPointerCapture(id);const r=el.getBoundingClientRect();lastA=Math.atan2(e.clientY-r.top-r.height/2,e.clientX-r.left-r.width/2);});
    el.addEventListener('pointermove',e=>{if(e.pointerId!==id)return;e.preventDefault();const r=el.getBoundingClientRect(),a=Math.atan2(e.clientY-r.top-r.height/2,e.clientX-r.left-r.width/2);let d=a-lastA;while(d>Math.PI)d-=2*Math.PI;while(d<-Math.PI)d+=2*Math.PI;lastA=a;const dd=deg(d);visual+=dd;arm.style.transform=`rotate(${visual}deg)`;if(kind==='az')state.azTarget=(state.azTarget+dd/6.5+3600)%360;else state.elTarget=clamp(state.elTarget+dd/8.5,15,80);updateNotebook();});
    const stop=e=>{if(id===null||e.pointerId!==id)return;id=null;};el.addEventListener('pointerup',stop);el.addEventListener('pointercancel',stop);
  }

  // Joystick
  const joyEl=$('joystick'),joyKnob=$('joyKnob');
  function setJoy(e){const r=joyEl.getBoundingClientRect(),cx=r.left+r.width/2,cy=r.top+r.height/2;let dx=e.clientX-cx,dy=e.clientY-cy,max=r.width*.35,l=Math.hypot(dx,dy)||1;if(l>max){dx*=max/l;dy*=max/l;}joy.x=dx/max;joy.y=dy/max;joyKnob.style.transform=`translate(${dx}px,${dy}px)`;}
  joyEl.addEventListener('pointerdown',e=>{e.preventDefault();touchUI();joyId=e.pointerId;joyEl.setPointerCapture(joyId);setJoy(e);});joyEl.addEventListener('pointermove',e=>{if(e.pointerId===joyId)setJoy(e)});function endJoy(e){if(joyId===null||e.pointerId!==joyId)return;joyId=null;joy={x:0,y:0};joyKnob.style.transform='translate(0,0)';}joyEl.addEventListener('pointerup',endJoy);joyEl.addEventListener('pointercancel',endJoy);
