  function updateSectorUnits(sec,dt){
    // Only detailed when the camera is nearby; distant fronts are abstracted in strategicTick.
    if(distance(sec,state.cam)>2600)return;
    for(const u of sec.units){
      u.cool-=dt;u.pop=Math.max(0,u.pop-dt);u.supp=Math.max(0,u.supp-dt*.12);
      const phase=Math.sin(state.time*.9+u.slot*1.7+(u.team==='enemy'?1.2:0));
      u.duck=clamp(u.supp>.5?.95:(phase>.45?.15:.78),0,1);
      if(u.cool<=0 && u.duck<.4){
        const foe=sec.units.filter(v=>v.team!==u.team)[Math.floor(rand(0,sec.units.filter(v=>v.team!==u.team).length))];
        if(foe){
          u.pop=.3;u.cool=u.role==='mg'?rand(.7,1.2):rand(1.5,2.7);const burst=u.role==='mg'?5:1;
          for(let i=0;i<burst;i++) state.tracers.push({x:u.x,y:u.y,x2:foe.x+rand(-18,18),y2:foe.y+rand(-18,18),life:.18+i*.025,max:.28,team:u.team});
          foe.supp=clamp(foe.supp+(u.role==='mg'?.22:.09),0,1);
        }
      }
    }
  }
